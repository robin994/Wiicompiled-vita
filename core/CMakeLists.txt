add_library(wiicompiled_core STATIC src/disc.cpp src/disc_index.cpp src/asset_archive.c)
target_include_directories(wiicompiled_core PUBLIC include)
target_compile_features(wiicompiled_core PUBLIC cxx_std_17)
target_compile_options(wiicompiled_core PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
option(WIC_ASSET_ZLIB "Read compressed cold native sidecars from PSARC" ON)
if(WIC_ASSET_ZLIB)
    target_compile_definitions(wiicompiled_core PRIVATE WIC_ASSET_ZLIB=1)
    if(VITA)
        target_link_libraries(wiicompiled_core PUBLIC z)
    else()
        find_package(ZLIB REQUIRED)
        target_link_libraries(wiicompiled_core PUBLIC ZLIB::ZLIB)
    endif()
endif()
if(VITA)
    target_compile_definitions(wiicompiled_core PRIVATE WIC_ASSET_VITA=1)
    target_link_libraries(wiicompiled_core PUBLIC crypto SceIofilemgr_stub SceLibKernel_stub pthread)
else()
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(wiicompiled_core PUBLIC OpenSSL::Crypto)
endif()
